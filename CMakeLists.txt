cmake_minimum_required(VERSION 3.30)
project(RDRX VERSION 0.1.0 LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC OR MSVC_VERSION LESS 1940 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "RDRX requires Windows x64 and MSVC 19.40 or newer")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
add_compile_definitions(WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
add_compile_options($<$<COMPILE_LANGUAGE:CXX>:/permissive-> $<$<COMPILE_LANGUAGE:CXX>:/EHsc> $<$<COMPILE_LANGUAGE:C,CXX>:/utf-8>)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
add_subdirectory(vendor/minhook EXCLUDE_FROM_ALL)
add_library(rdrx_core STATIC
 src/core.cpp src/memory.cpp src/discovery.cpp src/hooks.cpp src/engine.cpp
 src/plugins.cpp src/runtime.cpp src/crash.cpp src/live.cpp)
target_include_directories(rdrx_core PUBLIC include PRIVATE vendor/json)
target_include_directories(rdrx_core PUBLIC vendor/json)
target_link_libraries(rdrx_core PUBLIC minhook version psapi bcrypt dbghelp xinput9_1_0)
target_compile_options(rdrx_core PRIVATE /W4)
add_library(rdrx_sdk STATIC sdk/Core.cpp)
target_sources(rdrx_sdk PUBLIC FILE_SET CXX_MODULES BASE_DIRS sdk FILES
 sdk/RDRX.Core.ixx sdk/RDRX.Events.ixx sdk/RDRX.Input.ixx
 sdk/RDRX.Entities.ixx sdk/RDRX.World.ixx sdk/RDRX.ixx)
target_include_directories(rdrx_sdk PUBLIC include)
set_target_properties(rdrx_sdk PROPERTIES OUTPUT_NAME "RDRX.Core")
add_library(rdrx MODULE src/bootstrap.cpp)
target_link_libraries(rdrx PRIVATE rdrx_core)
set_target_properties(rdrx PROPERTIES PREFIX "" SUFFIX ".asi")
add_executable(rdrx_discover tools/discover.cpp)
target_link_libraries(rdrx_discover PRIVATE rdrx_core)
add_executable(rdrx_host tools/host.cpp)
target_link_libraries(rdrx_host PRIVATE rdrx_core)
add_library(imgui STATIC vendor/imgui/imgui.cpp vendor/imgui/imgui_draw.cpp
 vendor/imgui/imgui_tables.cpp vendor/imgui/imgui_widgets.cpp
 vendor/imgui/backends/imgui_impl_win32.cpp vendor/imgui/backends/imgui_impl_dx11.cpp)
target_include_directories(imgui PUBLIC vendor/imgui vendor/imgui/backends)
target_link_libraries(imgui PUBLIC d3d11 dxgi dwmapi)
add_library(rdrx_debug STATIC src/debug_ui.cpp)
target_link_libraries(rdrx_debug PUBLIC rdrx_core imgui)
add_executable(rdrx_dashboard WIN32 tools/dashboard.cpp)
target_link_libraries(rdrx_dashboard PRIVATE rdrx_debug)
enable_testing()
add_subdirectory(examples/Hello)
add_subdirectory(plugins/LocalCoop)
add_subdirectory(plugins/RenderTrace)
add_subdirectory(plugins/ControllerRoute)
add_subdirectory(plugins/VulkanOutput)
add_subdirectory(plugins/SplitScreen)
add_subdirectory(plugins/Online)
add_library(WorkerFixture MODULE tests/fixtures/Worker.cpp)
target_include_directories(WorkerFixture PRIVATE include)
set_target_properties(WorkerFixture PROPERTIES PREFIX "" SUFFIX ".SAAS")
add_custom_command(TARGET WorkerFixture POST_BUILD COMMAND ${CMAKE_COMMAND}
 "-DPLUGIN=$<TARGET_FILE:WorkerFixture>" "-DTEMPLATE=${PROJECT_SOURCE_DIR}/tests/fixtures/Worker.SAAS.json.in"
 -P "${PROJECT_SOURCE_DIR}/cmake/WriteManifest.cmake")
add_executable(rdrx_tests tests/tests.cpp tests/abi_c.c)
target_link_libraries(rdrx_tests PRIVATE rdrx_core rdrx_debug)
add_dependencies(rdrx_tests Hello WorkerFixture)
add_test(NAME rdrx_tests COMMAND rdrx_tests $<TARGET_FILE:Hello> "${PROJECT_SOURCE_DIR}" $<TARGET_FILE:WorkerFixture>)
add_test(NAME plugin_host COMMAND rdrx_host --self-test $<TARGET_FILE:Hello>)
add_test(NAME dashboard_graphics COMMAND rdrx_dashboard --smoke-test)
add_executable(rdrx_crash_probe tests/crash_probe.cpp)
target_link_libraries(rdrx_crash_probe PRIVATE rdrx_core)
add_test(NAME crash_diagnostics COMMAND ${CMAKE_COMMAND} "-DPROBE=$<TARGET_FILE:rdrx_crash_probe>"
 "-DBINARY_DIR=${CMAKE_BINARY_DIR}" -P "${PROJECT_SOURCE_DIR}/tests/check_crash.cmake")
install(TARGETS rdrx rdrx_discover rdrx_host rdrx_dashboard RUNTIME DESTINATION . LIBRARY DESTINATION .)
install(TARGETS Hello RUNTIME DESTINATION rdrx/plugins LIBRARY DESTINATION rdrx/plugins)
install(TARGETS rdrx_sdk ARCHIVE DESTINATION sdk/lib/$<CONFIG> FILE_SET CXX_MODULES DESTINATION sdk/modules CXX_MODULES_BMI DESTINATION sdk/ifc/$<CONFIG>)
install(TARGETS rdrx_debug ARCHIVE DESTINATION developer/lib/$<CONFIG>)
install(FILES include/rdrx/abi.h DESTINATION sdk/include/rdrx)
install(FILES cmake/RDRXConfig.cmake DESTINATION sdk/cmake)
install(DIRECTORY config/ DESTINATION rdrx/config)
install(DIRECTORY signatures/ DESTINATION rdrx/signatures)
install(DIRECTORY docs/ DESTINATION docs)
install(FILES README.md LICENSE THIRD_PARTY.md DESTINATION .)
install(DIRECTORY reports/ DESTINATION reports)
install(DIRECTORY examples/Hello/ DESTINATION examples/Hello)
install(FILES tools/console.ps1 DESTINATION tools)
install(FILES tools/coop-session.ps1 DESTINATION tools)
install(TARGETS LocalCoop ControllerRoute RenderTrace VulkanOutput RUNTIME DESTINATION experimental/plugins LIBRARY DESTINATION experimental/plugins)
install(FILES
 "$<TARGET_FILE:LocalCoop>.json" "$<TARGET_FILE:ControllerRoute>.json" "$<TARGET_FILE:RenderTrace>.json" "$<TARGET_FILE:VulkanOutput>.json"
 DESTINATION experimental/plugins)
install(FILES signatures/render-trace-1491.50.json DESTINATION experimental/plugins RENAME render-trace.json)
install(FILES signatures/vulkan-output-1491.50.json DESTINATION experimental/plugins RENAME vulkan-output.json)
install(FILES vendor/vulkan/LICENSE.txt DESTINATION licenses RENAME Vulkan-Headers.txt)
install(FILES vendor/minhook/LICENSE.txt DESTINATION licenses RENAME MinHook.txt)
install(FILES vendor/json/LICENSE.MIT DESTINATION licenses RENAME nlohmann-json.txt)
install(FILES vendor/imgui/LICENSE.txt DESTINATION licenses RENAME Dear-ImGui.txt)
